\section{De l'IA générative à l'agentique}
% ============================================================

\begin{frame}{Problème : chat $\neq$ agent}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{IA générative (chat)}
        \begin{itemize}
            \item Conversation, texte in / texte out
            \item Pas d'outils sur le dépôt
            \item Pas de boucle plan $\rightarrow$ verify
            \item Contexte = le prompt
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{IA agentique}
        \begin{itemize}
            \item Tool calling (shell, diff, lint, MCP\dots)
            \item Boucle : plan $\rightarrow$ act $\rightarrow$ verify
            \item Mémoire projet (rules, docs, lessons)
            \item Orchestration (worktrees, subagents)
        \end{itemize}
    \end{columns}
    \vspace{0.6cm}
    \begin{block}{Warning}
        Passer à l'agentique, c'est une nouvelle manière de travailler, il n'y a pu de code a proprement parler. 
        Ca demande beaucoup de comprehension du code généré et de la documentation tres bien construite.
        La charge cognitive est tres elevee si on ne controle pas bien ses agents.
        Il n'y a pu de moments chill ou on pisse des lignes de code. C'est l'agent qui code maintenant.
    \end{block}
\end{frame}

\section{Qu'est ce qu'un Harnais frugal}
% ============================================================

\begin{frame}{Harness : le cockpit de l'agent}
    \textbf{Ce que le harness apporte :}
    \begin{itemize}
        \item Tool calling, skills, rules, hooks, subagents, MCP
        \item Intégration IDE (diff, lint, terminal) ou orchestration multi-agents
    \end{itemize}
    \vspace{0.3cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{IDE-agents}
        \begin{itemize}
            \item Cursor, Windsurf, Continue\dots
            \item Éditeur fort : diff, lint, terminal
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{CLI / multi-agent}
        \begin{itemize}
            \item Claude Code, OpenCode, Codex\dots
            \item Parallèle, worktrees, manageur d'agents
        \end{itemize}
    \end{columns}
    \vspace{0.4cm}
    \begin{alertblock}{Point souvent oublié}
        Les benchmarks « modèle seul » sous-estiment le scaffolding.\\
        Selon l'outil choisi, l'expérience est \textbf{radicalement} différente.
    \end{alertblock}
\end{frame}

\begin{frame}{Preuve : Astra et le poids du harness}
    \begin{block}{ARC-AGI-3 --- même modèle, deux harness}
        Source : ARC Prize / OpenAI GPT-6 Astra
    \end{block}
    \vspace{0.3cm}
    \begin{table}
        \centering
        \begin{tabular}{lcc}
            \toprule
            \textbf{Harness} & \textbf{Score} \\
            \midrule
            Standard (neutre) & $\approx$\,62,7\,\%  \\
            Provider Adapter & $\approx$\,99,9\,\%  \\
            \bottomrule
        \end{tabular}
    \end{table}
    \vspace{0.4cm}
    \centering
    \Large
    Même modèle, même context, même tâche : écart énorme \\ 
    $\Rightarrow$ \textbf{le harness compte énormement}.
\end{frame}

\begin{frame}{Qu'est-ce qui fait un bon harness ?}
    \textbf{Capacités}
    \begin{itemize}
        \item Tool calling fiable (shell, diff, lint, MCP)
        \item Gestion d'état : raisonnement, compaction, mémoire de session
        \item Boucles de feedback dans l'outil (pas hors de l'éditeur)
        \item Extensibilité : skills, rules, hooks, subagents
    \end{itemize}
    \vspace{0.25cm}
    \textbf{Signaux d'un harness faible}
    \begin{itemize}
        \item Peu d'outils, ou outils mal branchés
        \item Pas de compaction $\Rightarrow$ contexte saturé
        \item Pas de verify (lint / tests) dans la boucle
        \item Coût élevé même sur une tâche simple
    \end{itemize}
    \vspace{0.35cm}
    \begin{block}{Critère pratique}
        Un bon harness \textbf{amplifie} le modèle.
        Un mauvais harness le \textbf{gaspille} (tokens, temps, crédits).
    \end{block}
\end{frame}

\section{Context \& amélioration continue}
% ============================================================

\begin{frame}{Context : la mémoire projet}
    \begin{itemize}
        \item \textbf{Bootstrap} --- injection initiale 
        \item \textbf{Rules} --- contraintes permanentes 
        \item \textbf{Lessons} --- mémoire opérationnelle inter-project
        \item \textbf{Router} --- quand lire quelle doc 
        \item \textbf{Docs spé} --- support agent + utilisateur pour une tâche précise
    \end{itemize}
    \vspace{0.4cm}
    \begin{block}{C'est ici que votre matière grise compte}
        C'est ici que vous allez passer le plus de temps, rédigez une \textbf{documentation de qualité} n'est pas facile. 
        A chaque fois que votre agent va divergé de votre intention initiale cest que votre documentation n'est pas assez bonne.
        De plus, si elle est trop longue, elle consomera trop. \\
        C'est comme si vous deviez redigerles spec de tous les ticket d'edithor d'un coup avec le moins de caractère que possible et que sa reste clair.
    \end{block}
\end{frame}

\section{Orchestration \& méthodes}
% ============================================================

\begin{frame}{Orchestration : le chantier}
    \begin{itemize}
        \item \textbf{1 agent = 1 worktree = 1 plan}
        \item Subagents pour \textbf{isoler} le contexte 
        \item intégration MCP --- TU/TI BE/FE et meme Tests manuelles
        \item Review via \textbf{\texttt{git diff}} --- pas de confiance aveugle
        \item code review -- bugbot (ex. codeRabbit)
    \end{itemize}
    \vspace{0.5cm}
    \begin{block}{Discipline}
        L'agent propose et exécute ; l'humain valide ce qui est irréversible
    \end{block}
\end{frame}

\begin{frame}{Méthodes (cadres, encore expérimental)}
    \begin{table}
        \centering
        \small
        \begin{tabular}{lp{10cm}}
            \toprule
            \textbf{Cadre} & \textbf{Angle} \\
            \midrule

\section{Provider, privacy, coûts}
% ============================================================

\begin{frame}{Provider : nécessaire}
    \begin{itemize}
        \item LLM plus ou moins fort : fenêtre de contexte, pricing, latence
        \item Critères utiles : qualité du \textbf{tool-calling}, rate limits, rétention data
        \item Cloud $\Rightarrow$ soumis au \textbf{paramétrage labo}
    \end{itemize}
    \vspace{0.4cm}
    \begin{alertblock}{Paradoxe}
        C'est l'étape du workflow la moins différenciante au-dela d'un seuil minimal de performance qui a été atteint depuis un moment par les plus gros labo (Sonnet 4.5, GLM 5.2 --- il y a 1 an).
    \end{alertblock}
    \vspace{0.3cm}
    \begin{block}{Message}
        On choisit un modèle \textbf{compatible avec le harness}, et dont le pricing n'est pas abusé. 
    \end{block}
\end{frame}

\begin{frame}{Privacy \& self-host}    
    \textbf{Outils}
    \begin{itemize}
        \item \textbf{Ollama} --- lancer et modifier un LLM sur sa machine
        \item \textbf{vLLM} --- serveur de LLM performant et multi-utilisateur  
        \item \textbf{Hugging Face} --- github des differents modèle
    \end{itemize}
    \textbf{Réalité}
    \begin{itemize}
        \item Hardware conséquent --- \textbf{Qwen3.6-27B} (Q4) $\approx$\,17\,Go VRAM ; \\
              $\Rightarrow$ GPU type \textbf{RTX 4090 24\,Go} (souvent 35--55\,tok/s)
        \item Modèle open-weights $\neq$\ open-source
        \item Tool-use local encore derrière le cloud top-tier
    \end{itemize}

    \vspace{0.5cm}
    \begin{block}{Cas d'usage}
        Code sensible, offline, personalisation modèles --- accepter le trade-off sur qualité/performance.
    \end{block}
\end{frame}

\section{Angles transverses \& conclusion}

        \begin{itemize}

            \item \textbf{Coût \& budgets} --- tokens / session

        \end{itemize}
        \column{0.48\textwidth}
        \begin{itemize}
            \item \textbf{Sécurité} --- injection, MCP, secrets
            \item \textbf{Tracabilité} --- transcripts, traces outils
            \item \textbf{Adaptation} --- owners vs users 
        \end{itemize}
    \end{columns}
    \vspace{0.6cm}
    \begin{block}{Fail encore d'actuallité}
        Mauvaise ou sur-utilisation d'outils $\cdot$ boucles infinies $\cdot$ contexte saturé $\cdot$ rules qui pourrissent  
    \end{block}
\end{frame}
